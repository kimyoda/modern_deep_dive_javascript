# 19.12 정적 프로퍼티/메서드

정적 프로퍼티와 정적 메서드는 인스턴스를 생성하지 않고도 생성자 함수 객체에서 직접 참조하거나 호출할 수 있는 멤버다.

```js
// 생성자 함수
function Person(name) {
  this.name = name;
}

// 프로토타입 메서드
Person.prototype.sayHello = function () {
  console.log(`Hi! My name is ${this.name}`);
};

// 정적 프로퍼티
Person.staticProp = "static prop";

// 정적 메서드
Person.staticMethod = function () {
  console.log("staticMethod");
};

const me = new Person("Lee");

// 생성자 함수에 추가한 정적 프로퍼티/메서드는 생성자 함수로 호출
Person.staticMethod(); // staticMethod

// 정적 프로퍼티/메서드는 생성자 함수가 생성한 인스턴스로 참조/호출할 수 없음
// 인스턴스로 참조/호출할 수 있는 프로퍼티/메서드는 프로토타입 체인 상에 존재해야 한다
try {
  me.staticMethod();
} catch (error) {
  console.log(error.name); // TypeError
}
```

`Person` 생성자 함수 객체가 직접 소유한 프로퍼티와 메서드를 정적 프로퍼티/메서드라고 한다. `Person`의 정적 멤버는 함수 객체 자체에, `sayHello`는 `Person.prototype`에 등록된다.

인스턴스는 자신의 프로토타입 체인에 있는 메서드를 사용할 수 있지만, 생성자 함수 객체 자체의 정적 메서드는 일반적으로 인스턴스에서 호출할 수 없다. 예를 들어 `Object.create`는 `Object`의 정적 메서드이므로 일반 객체의 `obj.create()`처럼 호출할 수 없다.

**예제 19-57**

```js
// Object.create는 정적 메서드다
const obj = Object.create({ name: "Kim" });

// Object.prototype.hasOwnProperty는 프로토타입 메서드
obj.hasOwnProperty("name"); // false
```

인스턴스 상태(`this`)를 사용하지 않는 기능은 정적 메서드로 설계할 수 있다. 다만 호출 위치가 달라지므로 기존 API의 사용 코드를 함께 수정해야 한다.

**예제 19-58**

```js
function Foo() {}

// 프로토타입 메서드
// this를 참조하지 않는 프로토타입 메서드는 정적 메서드로 변경해도 동일한 효과를 얻을 수 있음
Foo.prototype.x = function () {
  console.log("x");
};

const foo = new Foo();
// 프로토타입 메서드를 호출하면 인스턴스를 생성해야 한다
foo.x(); // x

// 정적 메서드
Foo.x = function () {
  console.log("x");
};

// 정적 메서드는 인스턴스를 생성하지 않아도 호출할 수 있음
Foo.x();
```
